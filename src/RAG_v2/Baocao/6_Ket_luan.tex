\section{Kết luận}

Trong khuôn khổ đồ án tốt nghiệp, sinh viên đã xây dựng thành công hệ thống RAG (Retrieval-Augmented Generation) Chatbot hỗ trợ sinh viên hỏi-đáp tự động về các quy định, quy chế đào tạo của Đại học Bách khoa Hà Nội. Hệ thống được thiết kế theo kiến trúc 8 tầng module hóa và đã giải quyết được các vấn đề chính sau:

\begin{itemize}
    \item Xây dựng pipeline xử lý PDF đa phương thức với Docling, PyMuPDF4LLM và olmOCR, đặc biệt giải quyết vấn đề font encoding cho văn bản tiếng Việt;
    
    \item Phát triển 4 loại Chunker chuyên biệt (OlmOcrLegalChunker, RecursiveChunker, STSVChunker, ArticleLevelLegalChunker) với khả năng nhận diện cấu trúc Chương-Điều-Khoản-Điểm, bảo toàn metadata và ngữ cảnh pháp lý;
    
    \item Triển khai Dual Embedding Pipeline với BGE-M3 và E5-Multilingual (1024 chiều), lưu trữ trên Qdrant Vector Database với dual named vectors;
    
    \item Triển khai Hybrid Search kết hợp Qdrant (Vector) + Elasticsearch (BM25) qua Reciprocal Rank Fusion, với cấu hình tối ưu v0.8/k0.2;
    
    \item Triển khai Cross-encoder Reranking với BAAI/bge-reranker-v2-m3 để cải thiện độ chính xác;
    
    \item Phát triển Query Router (dual-mode: classifier + LLM) và Query Reflection (LLM-based rewriting) cho xử lý truy vấn thông minh;
    
    \item Xây dựng GeminiLLM tích hợp Gemini 2.5 Flash qua OpenAI-compatible endpoint, với Self Evaluation tự đánh giá chất lượng câu trả lời;
    
    \item Tích hợp Tavily Web Search làm fallback và MongoDB Memory cho lịch sử hội thoại;
    
    \item Phát triển Backend API với FastAPI (SSE streaming) và Frontend với React 18 + TypeScript + Vite.
\end{itemize}

\section{Bài học kinh nghiệm (Lessons Learned)}

\subsection{Về xử lý PDF tiếng Việt}
\begin{itemize}
    \item Font encoding là vấn đề phổ biến với PDF Việt Nam cũ - cần có fallback mechanism
    \item Vision-based OCR (olmOCR) là giải pháp cuối cùng đáng tin cậy
    \item Table extraction cần xử lý riêng, không nên dựa vào text extraction thông thường
\end{itemize}

\subsection{Về Retrieval}
\begin{itemize}
    \item Semantic search đơn thuần không đủ cho văn bản pháp lý - cần kết hợp BM25
    \item Truy vấn phủ định tiếng Việt (``không được xét'') cần xử lý đặc biệt
    \item RRF fusion hiệu quả hơn weighted sum vì không cần normalize điểm
\end{itemize}

\subsection{Về Chunking}
\begin{itemize}
    \item Bảo toàn cấu trúc Chương-Điều-Khoản quan trọng cho trích dẫn
    \item Parent-child chunk hierarchy giúp context đầy đủ hơn
    \item Metadata (source file, article, clause) cần thiết cho transparency
\end{itemize}

\section{Hạn chế còn tồn tại}

\begin{itemize}
    \item Chưa có query caching (embedding cache, retrieval cache, response cache), mỗi query phải embedding lại;
    \item Chưa xử lý phạm vi hiệu lực của văn bản: chưa filter theo khóa sinh viên, thời điểm áp dụng;
    \item Chưa có cơ chế nhận diện Student Context (khóa, ngành) để cá nhân hóa câu trả lời;
    \item Chưa giải quyết cross-reference giữa các văn bản (``theo quy định tại Điều X'');
    \item Chưa xử lý document override (văn bản mới thay thế văn bản cũ);
    \item Chưa có cơ chế update tài liệu tự động khi có quy chế mới;
    \item Cross-encoder reranking làm tăng latency ($\sim$200ms thêm cho 20 documents).
\end{itemize}

\section{Hướng phát triển trong tương lai}

\textit{Lưu ý: Giao diện chatbot hiện tại chỉ là demo để kiểm tra khả năng retrieval của hệ thống. Mục tiêu chính của đề tài là xây dựng hệ thống tự động trả lời email tư vấn học tập cho sinh viên.}

\subsection{Phase 1: Hoàn thiện hệ thống RAG (1--2 tháng)}
\begin{itemize}
    \item \textbf{[HOÀN THÀNH]} Build evaluation dataset với 100+ test queries $\rightarrow$ \textit{2.390 câu hỏi trên 25 bộ dữ liệu}
    \item \textbf{[HOÀN THÀNH]} Hybrid search kết hợp BM25 + semantic retrieval $\rightarrow$ \textit{Hit@5 = 86.4\%}
    \item \textbf{[HOÀN THÀNH]} Re-ranking với cross-encoder để tăng precision
    \item \textbf{[HOÀN THÀNH]} Query Router dual-mode (classifier + LLM) + Query Reflection
    \item \textbf{[HOÀN THÀNH]} Self Evaluation + Tavily fallback + MongoDB Memory
    \item \textbf{[HOÀN THÀNH]} FastAPI Backend + React Frontend
    \item Implement query caching (semantic + exact match) để giảm latency
    \item Metadata-aware retrieval: filter theo khóa, ngành, thời điểm áp dụng
    \item Nâng cấp Agentic RAG: Agent tự lập kế hoạch, phân rã câu hỏi phức hợp
\end{itemize}

\subsection{Phase 2: Xây dựng Email Processing Pipeline (2-3 tháng)}
\begin{itemize}
    \item \textbf{Email Parser:} Tích hợp IMAP/POP3 để đọc email từ hộp thư phòng đào tạo
    \item \textbf{Intent Classification:} Phân loại email theo chủ đề (học bổng, tốt nghiệp, ngoại ngữ, đăng ký môn...)
    \item \textbf{Entity Extraction:} Trích xuất thông tin sinh viên (MSSV, họ tên, khóa, chương trình)
    \item \textbf{Context Enrichment:} Bổ sung context từ hồ sơ sinh viên nếu có quyền truy cập
    \item \textbf{Response Generation:} Sinh email trả lời với format chuẩn, trích dẫn điều khoản
    \item \textbf{Draft Review System:} Cán bộ review và approve trước khi gửi
\end{itemize}

\subsection{Phase 3: Triển khai Email Bot (3-4 tháng)}
\begin{itemize}
    \item \textbf{Email Sender:} Tích hợp SMTP để gửi email trả lời tự động
    \item \textbf{Template System:} Hệ thống template email theo từng loại câu hỏi
    \item \textbf{Signature Management:} Quản lý chữ ký email của phòng đào tạo
    \item \textbf{CC/Forward Logic:} Tự động forward email phức tạp cho cán bộ phụ trách
    \item \textbf{Response Time SLA:} Đảm bảo thời gian phản hồi < 30 phút trong giờ hành chính
\end{itemize}

\subsection{Phase 4: Giám sát và cải thiện (Liên tục)}
\begin{itemize}
    \item \textbf{Quality Monitoring:} Dashboard theo dõi chất lượng câu trả lời
    \item \textbf{Feedback Loop:} Thu thập phản hồi từ sinh viên và cán bộ
    \item \textbf{Escalation Rules:} Quy tắc chuyển tiếp email cần xử lý thủ công
    \item \textbf{Knowledge Base Update:} Cập nhật quy chế mới khi có thay đổi
    \item \textbf{Analytics:} Thống kê các câu hỏi thường gặp (FAQ) để cải thiện tài liệu
\end{itemize}

\section{Kết luận cuối cùng}

Tổng kết lại, đồ án ``\textbf{Nghiên cứu và đề xuất mô hình tự động trả lời email tư vấn học tập cho sinh viên}'' đã hoàn thành giai đoạn đầu tiên --- xây dựng và đánh giá hệ thống RAG hoàn chỉnh 8 tầng làm nền tảng cho email bot. Các đóng góp chính:

\begin{enumerate}
    \item \textbf{Pipeline xử lý PDF đa phương thức} giải quyết vấn đề font encoding tiếng Việt, xử lý thành công 137 tài liệu từ 3 nguồn dữ liệu
    \item \textbf{4 loại Chunker chuyên biệt cho văn bản pháp lý} bảo toàn cấu trúc Chương-Điều-Khoản, tạo 2.752 chunks chất lượng cao
    \item \textbf{Dual Embedding (BGE-M3 + E5)} lưu trữ trên Qdrant với dual named vectors
    \item \textbf{Hybrid Search với RRF fusion} kết hợp Qdrant + Elasticsearch, đạt Hit@5 = 86.4\% trên 2.390 queries
    \item \textbf{Cross-encoder reranking} với BAAI/bge-reranker-v2-m3 cải thiện độ chính xác
    \item \textbf{Query Router dual-mode + Reflection} xử lý truy vấn thông minh, context-aware
    \item \textbf{GeminiLLM + Self Evaluation} sinh câu trả lời và tự đánh giá chất lượng
    \item \textbf{Evaluation framework} quy mô lớn với 25 bộ dữ liệu, 2.390 câu hỏi
    \item \textbf{Full-stack deployment}: FastAPI backend (SSE streaming) + React frontend + Docker
\end{enumerate}

Hệ thống RAG đã chứng minh khả năng truy xuất chính xác thông tin từ văn bản quy chế đào tạo, với kiến trúc module hóa cho phép dễ dàng mở rộng và tích hợp. Đây là tiền đề quan trọng để phát triển hệ thống tự động trả lời email tư vấn học tập trong các giai đoạn tiếp theo.
