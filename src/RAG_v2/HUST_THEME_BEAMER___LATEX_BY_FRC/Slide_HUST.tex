	\section{Đóng góp nổi bật}

				\begin{frame}{1. Định tuyến và xử lý đa bước}
					\begin{columns}[T,onlytextwidth]
						\begin{column}{0.48\textwidth}
							{\bfseries\color{hustprimary} Chọn đúng đường xử lý}
							\vspace{0.08cm}

							\small
							\begin{itemize}
								\setlength{\itemsep}{0.12cm}
								\item Trò chuyện thường hoặc ngoài phạm vi.
								\item Simple RAG cho câu hỏi rõ nguồn.
								\item RAG đa bước cho câu hỏi phức tạp, so sánh, đa nguồn.
								\item Nếu bộ định tuyến thiếu tự tin: mở rộng miền dữ liệu hoặc dùng luồng dự phòng.
							\end{itemize}
						\end{column}
						\begin{column}{0.48\textwidth}
							{\bfseries\color{hustprimary} Bộ định tuyến ba tầng}
							\vspace{0.08cm}

							\small
							\begin{itemize}
								\setlength{\itemsep}{0.12cm}
								\item Regex xử lý nhanh trường hợp rõ ràng.
								\item ML chọn ý định: \texttt{blocked}, \texttt{chitchat}, \texttt{tool\_search}, \texttt{rag}.
								\item Nếu là \texttt{rag}: ML chọn miền dữ liệu \texttt{ctdt}, \texttt{quydinh}, \texttt{kehoach}, \texttt{stsv}.
								\item LLM chỉ đánh giá độ phức tạp khi câu hỏi chạm nhiều miền.
							\end{itemize}
						\end{column}
					\end{columns}

					\vspace{0.22cm}
					{\bfseries\color{hustprimary} Planner--Executor đa bước}

					\vspace{0.02cm}
					\small Với câu hỏi nhiều bước, Planner phân rã truy vấn, Executor truy hồi theo từng miền, Synthesizer tổng hợp câu trả lời cuối; nếu lỗi thì fallback về Simple RAG.
				\end{frame}

				\begin{frame}{2. Truy hồi lai}
			\begin{columns}[T,onlytextwidth]
				\begin{column}{0.48\textwidth}
					\begin{block}{Nguồn truy hồi}
						\scriptsize
						\begin{itemize}
								\item Vector search trên Qdrant.
								\item \textbf{BM25 Elasticsearch}: bộ phân tích tiếng Việt CocCoc.
								\item Hai nguồn bổ trợ nhau: khớp ngữ nghĩa và khớp chính xác theo mã/ngày/thuật ngữ học vụ.
						\end{itemize}
					\end{block}
				\end{column}
				\begin{column}{0.48\textwidth}
					\begin{block}{RRF}
						\scriptsize
						\begin{itemize}
							\item Hợp nhất danh sách vector và keyword theo \textbf{thứ hạng}, không nội suy điểm thô.
							\item Điểm ưu tiên độ mới cho \texttt{kehoach} là tín hiệu phụ sau hợp nhất: tối đa $+0.05$, giảm dần tới 365 ngày.
						\end{itemize}
					\end{block}
				\end{column}
			\end{columns}

			\vspace{0.06cm}
				{\scriptsize\textbf{Sau hợp nhất:} rerank cross-encoder $\rightarrow$ HyDE khi rerank yếu $\rightarrow$ mở rộng cha-con $\rightarrow$ lọc hiệu lực.}

			\vspace{0.04cm}
			{\scriptsize\textbf{Tìm kiếm web:} Tavily khi thiếu nguồn/thông tin mới; giới hạn tên miền chính thức và đưa vào ngữ cảnh kèm URL.}
		\end{frame}

		\begin{frame}{3. Truy hồi nhận biết hồ sơ}
			\begin{columns}[T,onlytextwidth]
				\begin{column}{0.48\textwidth}
					\begin{block}{Ánh xạ theo chương trình}
						\scriptsize
						\begin{itemize}
							\setlength{\itemsep}{0.04cm}
							\item Phân biệt ngành của sinh viên và ngành được hỏi.
							\item Tra danh mục học phần theo CTĐT: ``Mạng máy tính'' $\rightarrow$ \texttt{IT3080} (IT-E6) hoặc \texttt{IT3080E} (IT-E7).
							\item Giữ mã môn trong câu hỏi truy hồi để tránh lấy nhầm CTĐT.
						\end{itemize}
					\end{block}
				\end{column}
				\begin{column}{0.48\textwidth}
					\begin{block}{Dùng hồ sơ có kiểm soát}
						\scriptsize
						\begin{itemize}
							\setlength{\itemsep}{0.04cm}
							\item Chỉ đưa hồ sơ vào khi câu hỏi phụ thuộc ngành/khóa; câu hỏi chung không bị lẫn ngữ cảnh cá nhân.
							\item Nếu câu hỏi nêu ngành cụ thể, ưu tiên ngành được hỏi thay vì ngành của người hỏi.
							\item Với văn bản quy định, giữ thông tin ngành khi tài liệu không luôn gắn mã ngành trực tiếp.
						\end{itemize}
					\end{block}
				\end{column}
			\end{columns}

			\vspace{0.04cm}
			{\color{hustprimary}\bfseries\scriptsize Lọc theo ngữ cảnh trước khi tìm kiếm:}
			{\tiny CTĐT theo mã/tên ngành; quy định theo khóa áp dụng; kế hoạch theo tháng, năm hoặc độ mới.}

			\end{columns}
