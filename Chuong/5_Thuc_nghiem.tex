\subsection{Phân tích chi tiết}

\subsubsection{PyMuPDF4LLM}
\begin{itemize}
    \item \textbf{Ưu điểm:} Tốc độ nhanh nhất, output markdown tốt
    \item \textbf{Nhược điểm:} Fail với PDF có lỗi font encoding
    \item \textbf{Use case:} PDF có font embedding chuẩn
\end{itemize}

\subsubsection{Docling}
\begin{itemize}
    \item \textbf{Ưu điểm:} Structure detection xuất sắc, table extraction tốt
    \item \textbf{Nhược điểm:} Chậm hơn PyMuPDF, cũng fail với lỗi font
    \item \textbf{Use case:} PDF phức tạp với nhiều bảng biểu
\end{itemize}

\subsubsection{olmOCR}
\begin{itemize}
    \item \textbf{Ưu điểm:} Vision-based, không phụ thuộc font, 100\% success
    \item \textbf{Nhược điểm:} Chậm nhất, cần GPU
    \item \textbf{Use case:} PDF scan hoặc lỗi font encoding
\end{itemize}

\section{Đánh giá chất lượng câu trả lời}

\subsection{Ưu điểm}
\begin{itemize}
    \item Câu trả lời chính xác, dựa trên nội dung quy chế thực tế
    \item Trích dẫn nguồn cụ thể (Điều, Khoản)
    \item Ngôn ngữ rõ ràng, phù hợp với ngữ cảnh học vụ
    \item Retrieved documents có relevance score cao
\end{itemize}

\subsection{Nhược điểm và hạn chế}
\begin{itemize}
    \item Chưa có query caching - mỗi query phải embedding lại
    \item Single-stage retrieval - chưa có re-ranking
    \item Chưa tận dụng lịch sử hội thoại
    \item Mỗi query xử lý độc lập, không hỗ trợ follow-up questions
\end{itemize}

\section{Gợi ý cải tiến}

\subsection{Performance Optimization}
\begin{itemize}
    \item Implement query caching (semantic + exact match)
    \item Upgrade FAISS index từ Flat sang IVF cho scale lớn hơn
    \item Add conversation history management
\end{itemize}

\subsection{Retrieval Enhancement}
\begin{itemize}
    \item Hybrid search: BM25 + semantic
    \item Query rewriting với context
    \item Re-ranking với cross-encoder
\end{itemize}

\subsection{Generation Improvement}
\begin{itemize}
    \item Multi-document reasoning
    \item Table-aware response generation
    \item User feedback mechanism
\end{itemize}
